# ---------------------------------------------------------------------------
# OpenMote Gen 3.3 — Arduino board definition
#
# This is not a whole platform. It reuses the core, toolchain and build recipes
# from "esp32 by Espressif Systems" (install that through Boards Manager first)
# and only supplies this board's own defaults and pin variant.
#
# The point of it: every setting that the manual ESP32S3 Dev Module recipe asks
# you to get right is baked in below. In particular the flash/PSRAM combination,
# which is the one that silently stops the board booting if you pick "OPI 80MHz"
# by mistake.
#
# Verified against esp32 core 3.3.7.
# ---------------------------------------------------------------------------

# -- Chip / core ----------------------------------------------------------

openmote_gen3.build.variant=openmote_gen3
openmote_gen3.build.board=OPENMOTE_GEN3
openmote_gen3.build.f_cpu=240000000L

# -- The settings that used to be a checklist -----------------------------
# ESP32-S3-WROOM-1-N16R8: 16 MB QUAD flash + 8 MB OCTAL PSRAM.
# memory_type resolves to {boot}_{psram_type} = qio_opi, which is the only
# correct combination for this module. Selecting "OPI 80MHz" as the flash mode
# in the generic Dev Module recipe yields opi_opi and the board will not boot;
# here it simply is not offered.
openmote_gen3.build.flash_size=16MB
openmote_gen3.build.flash_freq=80m
openmote_gen3.build.flash_mode=dio
openmote_gen3.build.boot=qio
openmote_gen3.build.boot_freq=80m
openmote_gen3.build.psram_type=opi
openmote_gen3.build.memory_type={build.boot}_{build.psram_type}
openmote_gen3.build.extra_flags=-DBOARD_HAS_PSRAM

# USB CDC on boot is mandatory, not a preference: GPIO43/44 (UART0) are wired
# to D-pad Up and Down, so USB CDC is the only usable console on this board.
openmote_gen3.build.cdc_on_boot=1
openmote_gen3.build.msc_on_boot=0
openmote_gen3.build.dfu_on_boot=0
openmote_gen3.build.usb_mode=1

openmote_gen3.build.partitions=app3M_fat9M_16MB
openmote_gen3.build.defines=
openmote_gen3.build.loop_core=
openmote_gen3.build.event_core=
openmote_gen3.build.code_debug=0
